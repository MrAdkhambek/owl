# owl::redis -- async Redis over coro and hiredis, header-only, in sql's
# layout: public headers under include/redis/, consumed with their "redis/"
# prefix, the FILE_SET as the only header list, and the same tree /
# manifest / umbrella guard so the three cannot drift.
#
# hiredis comes from pkg-config, the way owl finds libh2o. Note that
# libh2o-evloop.a bundles its own hiredis 1.2.0 objects for h2o's
# h2o_redis_client_t and installs no headers for them; those archive members
# are linked only if something references h2o_redis_* or includes
# h2o/redis.h / h2o/hiredis_.h. Nothing in owl may do either while this
# target links brew's hiredis: two hiredis implementations in one process
# would allocate replies in one and free them in the other.

add_library(redis INTERFACE)
add_library(${PROJECT_NAME}::redis ALIAS redis)

target_compile_features(redis INTERFACE cxx_std_23)

target_include_directories(redis INTERFACE
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)

target_sources(redis INTERFACE
        FILE_SET HEADERS
        BASE_DIRS "${CMAKE_CURRENT_SOURCE_DIR}/include"
        FILES
        include/redis/redis.h
        include/redis/error.h
        include/redis/reply.h
        include/redis/config.h
        include/redis/connection.h
        include/redis/detail/deadline.h
        include/redis/client.h
        include/redis/command.h
        include/redis/commands.h
        include/redis/pubsub.h
        include/redis/detail/args.h
)

find_package(PkgConfig REQUIRED)
pkg_check_modules(HIREDIS REQUIRED IMPORTED_TARGET hiredis)
target_link_libraries(redis INTERFACE coro PkgConfig::HIREDIS)
target_compile_definitions(redis INTERFACE OWL_ENABLE_REDIS)

# Tree / manifest / umbrella consistency: see cmake/owl_header_guard.cmake.
owl_header_guard(redis redis _redis_public)

if (BUILD_TESTING)
    # One TU per public header, each including only that header, so a
    # header that forgot an include cannot hide behind the umbrella.
    set(_redis_hdr_dir "${CMAKE_CURRENT_BINARY_DIR}/header_checks")
    set(_redis_hdr_srcs "")
    foreach (_h IN LISTS _redis_public)
        string(REPLACE "/" "_" _stem "${_h}")
        string(REPLACE ".h" "" _stem "${_stem}")
        set(_src "${_redis_hdr_dir}/${_stem}.cpp")
        file(CONFIGURE OUTPUT "${_src}"
                CONTENT "// generated by redis/CMakeLists.txt -- do not edit\n#include <${_h}>\n")
        list(APPEND _redis_hdr_srcs "${_src}")
    endforeach ()

    add_library(redis_headers OBJECT ${_redis_hdr_srcs})
    target_link_libraries(redis_headers PRIVATE ${PROJECT_NAME}::redis)

    # Mirrors the header tree: tests/live/test_live -> target redis_live_test_live.
    function(redis_add_test path)
        string(REPLACE "/" "_" name "${path}")
        set(name "redis_${name}")
        add_executable(${name} "${CMAKE_CURRENT_SOURCE_DIR}/tests/${path}.cpp")
        target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
        target_link_libraries(${name}
                PRIVATE
                ${PROJECT_NAME}::redis
                GTest::gtest_main)
        add_test(NAME ${name} COMMAND ${name})
    endfunction()

    # Misuse that must not compile. EXCLUDE_FROM_ALL so the normal build
    # never touches it; ctest builds it and expects the failure.
    function(redis_add_compile_fail_test name)
        add_executable(${name} EXCLUDE_FROM_ALL "${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_fail/${name}.cpp")
        target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tests")
        target_link_libraries(${name} PRIVATE ${PROJECT_NAME}::redis)
        add_test(NAME ${name}
                COMMAND ${CMAKE_COMMAND} --build "${CMAKE_BINARY_DIR}"
                        --target ${name} --config $<CONFIG>)
        set_tests_properties(${name} PROPERTIES WILL_FAIL TRUE)
    endfunction()

    redis_add_test(test_error)
    redis_add_test(test_reply)
    redis_add_test(test_connection)
    redis_add_test(test_client)
    redis_add_test(test_args)
    redis_add_test(test_command)
    redis_add_test(test_commands)
    redis_add_test(test_pubsub)
    redis_add_test(live/test_live)
    redis_add_compile_fail_test(rejects_unbindable_redis_argument)
endif ()
